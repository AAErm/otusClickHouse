```sql
USE restaurant;
CREATE TABLE orders  
(  
    order_id UInt32 COMMENT 'Уникальный идентификатор заказа',  
    table_number UInt32 COMMENT 'Номер столика',  
    order_date DateTime DEFAULT now() COMMENT 'Дата и время заказа',  
    total_amount Nullable(Float64) COMMENT 'Общая сумма заказа',  
    status LowCardinality(String) COMMENT 'Статус заказа (например, "Новый", "В обработке", "Доставлен", "Отменен", "Завершен")',  
    items_count Nullable(UInt32) COMMENT 'Количество позиций в заказе'  
)   
ENGINE = MergeTree()  
ORDER BY order_id  
SETTINGS index_granularity = 8192;  
```

6. Создал таблицу меню ресторана
```sql
-- Создаем таблицу "Меню ресторана"  
CREATE TABLE menu (  
    menu_id UInt32, -- Уникальный идентификатор блюда  
    name LowCardinality(String), -- Название блюда (LowCardinality для оптимизации хранения строк)  
    category LowCardinality(String), -- Категория блюда (например, "Супы", "Десерты")  
    price Decimal(10, 2), -- Цена блюда с двумя знаками после запятой  
    status Enum8('Available' = 1, 'Out of Stock' = 2, 'Seasonal' = 3), -- Статус доступности блюда  
    calories Nullable(UInt16), -- Калорийность блюда (Nullable, так как не для всех блюд это указано)  
    ingredients Array(String), -- Список ингредиентов блюда  
    created_at DateTime DEFAULT now() -- Дата и время добавления записи (по умолчанию текущее время)  
)   
ENGINE = MergeTree()  
ORDER BY menu_id; -- Указываем сортировку по уникальному идентификатору  
```
